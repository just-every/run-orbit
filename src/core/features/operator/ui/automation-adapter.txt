import {createServer} from 'node:http';
import {createHmac,timingSafeEqual} from 'node:crypto';
const target=new URL(orbitTarget);
if(target.username||target.password||target.search||target.hash||!/^\/api\/v1\/signals\/[a-zA-Z0-9_-]{1,80}$/.test(target.pathname)||!(target.protocol==='https:'||(process.env.ORBIT_ADAPTER_ALLOW_LOOPBACK==='1'&&target.protocol==='http:'&&['localhost','127.0.0.1','[::1]'].includes(target.hostname))))throw Error('Use the fixed HTTPS Orbit endpoint. Explicit loopback mode is for local development only.');
const secret=process.env.ORBIT_SOURCE_SECRET;
const credential=process.env.ORBIT_ADAPTER_TOKEN;
if(!secret||!credential||credential.length<32||credential===secret)throw Error('Configure distinct private ORBIT_SOURCE_SECRET and ORBIT_ADAPTER_TOKEN (at least 32 random characters).');
const port=Number(process.env.PORT??8789);
if(!Number.isInteger(port)||port<1||port>65535)throw Error('Invalid PORT.');
const authenticated=value=>{
 const expected=Buffer.from('Bearer '+credential),actual=Buffer.from(value??'');
 return expected.length===actual.length&&timingSafeEqual(expected,actual);
};
function body(request){return new Promise((resolve,reject)=>{
 const chunks=[];let size=0;
 request.on('data',chunk=>{size+=chunk.length;if(size>262144){request.pause();reject(Object.assign(Error('Event exceeds 262144 bytes.'),{status:413}));}else chunks.push(chunk);});
 request.on('end',()=>resolve(Buffer.concat(chunks)));
 request.on('error',reject);request.on('aborted',()=>reject(Error('Request interrupted.')));
});}
const server=createServer(async(request,response)=>{
 const reply=(status,value,contentType='application/json')=>{response.writeHead(status,{...(contentType?{'Content-Type':contentType}:{}),'Cache-Control':'no-store','Connection':'close'});response.end(value,()=>request.destroy());};
 if(request.method!=='POST'||request.url!=='/events')return reply(404,JSON.stringify({error:'Use POST /events.'}));
 if(!authenticated(request.headers.authorization))return reply(401,JSON.stringify({error:'Adapter authentication required.'}));
 if(!/^application\/json(?:;|$)/i.test(request.headers['content-type']??''))return reply(415,JSON.stringify({error:'Send application/json.'}));
 let raw;
 try{
  raw=new TextDecoder('utf-8',{fatal:true}).decode(await body(request));
  const event=eventSchema.strict().safeParse(JSON.parse(raw));
  if(!event.success||event.data.type!=='form.submitted'||!event.data.email||Date.parse(event.data.occurredAt)>Date.now())return reply(400,JSON.stringify({error:'Send a form.submitted event with its original eventId, occurredAt, subject and customer email. Preserve the complete original body on retries.'}));
 }catch(error){return reply(error.status??400,JSON.stringify({error:error.status===413?'Event exceeds 262144 bytes.':'Invalid event body.'}));}
 // Only delivery headers change. Never mint or rewrite event identity or content.
 const timestamp=String(Math.floor(Date.now()/1000));
 const signature=createHmac('sha256',secret).update(timestamp+'.'+raw).digest('hex');
 try{
  const upstream=await fetch(orbitTarget,{method:'POST',redirect:'error',signal:AbortSignal.timeout(10000),headers:{'Content-Type':'application/json','X-Orbit-Timestamp':timestamp,'X-Orbit-Signature':signature},body:raw});
  return reply(upstream.status,Buffer.from(await upstream.arrayBuffer()),upstream.headers.get('content-type'));
 }catch{return reply(502,JSON.stringify({error:'Orbit delivery outcome is unknown. Retry the same exact original body; do not create a new event.'}));}
});
server.requestTimeout=15000;server.headersTimeout=10000;
server.listen(port,process.env.HOST??'127.0.0.1',()=>console.log('ADAPTER_READY'));
for(const signal of ['SIGINT','SIGTERM'])process.on(signal,()=>{server.close();server.closeAllConnections();});

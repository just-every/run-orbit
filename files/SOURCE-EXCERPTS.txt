11: export function useOperatorResource<T>(key: string, load: () => Promise<T>, initialData?: T): ResourceState<T> {
12:   const [data, setData] = useState<T | null>(initialData ?? null);
13:   const [failure, setFailure] = useState<{ key: string; error: Error } | null>(null);
14:   const error = failure?.key === key ? failure.error : null;
15:   const [request, setRequest] = useState(0);
16: 
17:   useEffect(() => {
18:     let active = true;
19:     // Retain the server-rendered result while refreshing this view.
20:     setFailure(null);
21: 
22:     load().then(
23:       (next) => { if (active) setData(next); },
24:       (next: unknown) => { if (active) setFailure({ key, error: next instanceof Error ? next : new Error("Orbit could not load this view.") }); }
25:     );
26: 
27:     return () => { active = false; };
28:   }, [key, load, request]);
29: 
30:   // The main detail read owns the route's recovery state; nested panels keep their own errors.
31:   if (error && (error as Error & { status?: number }).status === 404 && /^(opportunity|contact|company|draft|meeting|assistance):/.test(key)) {
32:     throw notFound();
33:   }
34: 
35:   return { data, error, loading: !data && !error, reload: () => setRequest((current) => current + 1) };
